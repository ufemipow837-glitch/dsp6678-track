%% ====================== 读取实测数据 ======================
fprintf('正在读取实测数据: %s\n', csv_file);

% 使用textscan读取CSV（兼容R2017b）
fid = fopen(csv_file, 'r');
if fid == -1
    error('无法打开文件: %s', csv_file);
end

% 跳过表头行（中文表头直接跳过，不解析内容避免编码问题）
fgetl(fid);

% 读取数据列（按列位置读取，不依赖中文列名）
data = textscan(fid, '%d %d %d %f %f %f %f %f %f %d %f %d %d %d %d', ...
    'Delimiter', ',', 'TreatAsEmpty', 'NA');
fclose(fid);

% 提取各列（列号对应CSV第1-15列）
idx_all       = data{1};   % 序号
beam_all      = data{2};   % 波位号
t_beidou_all  = data{3};   % 北斗时间戳
t_ms_all      = data{4};   % 时间戳(ms)
az_all        = data{5};   % 方位(度)
r_all         = data{6};   % 距离(m)
el_all        = data{7};   % 俯仰(度)
h_all         = data{8};   % 高度(m)
v_all         = data{9};   % 径向速度(m/s)
snr_all       = data{10};  % 信噪比
rcs_all       = data{11};  % RCS
pulse_all     = data{12};  % 宽窄脉冲
el_beam_all   = data{13};  % 俯仰波束
targets_all   = data{14};  % 本波位目标数
valid_all     = data{15};  % 是否有效点

% 筛选有效点 + 无人机特征
drone_mask = (valid_all == 1) & ...
             (r_all > R_MIN) & (r_all < R_MAX) & ...
             (abs(v_all) > V_MIN) & (abs(v_all) < V_MAX) & ...
             (el_all > EL_MIN) & ...
             (az_all > AZ_MIN) & (az_all < AZ_MAX);

t_meas    = t_ms_all(drone_mask);
az_meas   = az_all(drone_mask);
el_meas   = el_all(drone_mask);
r_meas    = r_all(drone_mask);
v_meas    = abs(v_all(drone_mask));
beam_meas = beam_all(drone_mask);

% 所有有效点（用于3D全景图中的杂波显示）
valid_mask = (valid_all == 1) & (r_all > 0);
t_all_valid = t_ms_all(valid_mask);
az_all_valid = az_all(valid_mask);
el_all_valid = el_all(valid_mask);
r_all_valid  = r_all(valid_mask);

% 按时间排序
[t_meas, sort_idx] = sort(t_meas);
az_meas  = az_meas(sort_idx);
el_meas  = el_meas(sort_idx);
r_meas   = r_meas(sort_idx);
v_meas   = v_meas(sort_idx);

fprintf('  - 原始总点数: %d\n', length(t_ms_all));
fprintf('  - 有效点迹数: %d\n', length(t_all_valid));
fprintf('  - 筛选后无人机点数: %d\n', length(t_meas));

%% ====================== 提取跟踪结果 ======================
t_track  = track_results(:, 2);
az_track = track_results(:, 3);
el_track = track_results(:, 4);
r_track  = track_results(:, 5);
v_track  = track_results(:, 6);

fprintf('  - 跟踪结果点数: %d\n', length(t_track));
fprintf('  - 跟踪起始时间: %.0f ms (Frame %d)\n', t_track(1), track_results(1,1));
fprintf('  - 跟踪结束时间: %.0f ms (Frame %d)\n', t_track(end), track_results(end,1));

%% ====================== 时间对齐与误差计算 ======================
% 找共同时间点进行逐点对比
[common_t, idx_meas, idx_track] = intersect(t_meas, t_track);

if isempty(common_t)
    warning('没有找到共同时间点！将显示各自数据点，不计算逐点误差。');
    err_az = []; err_el = []; err_r = []; err_v = [];
else
    az_meas_match = az_meas(idx_meas);
    el_meas_match = el_meas(idx_meas);
    r_meas_match  = r_meas(idx_meas);
    v_meas_match  = v_meas(idx_meas);
    
    az_track_match = az_track(idx_track);
    el_track_match = el_track(idx_track);
    r_track_match  = r_track(idx_track);
    v_track_match  = v_track(idx_track);
    
    err_az = az_track_match - az_meas_match;
    err_el = el_track_match - el_meas_match;
    err_r  = r_track_match  - r_meas_match;
    err_v  = v_track_match  - v_meas_match;
end

%% ====================== 误差统计输出 ======================

fprintf('============================================================\n');
fprintf('           跟踪精度统计报告（有量测更新的帧）\n');

fprintf('匹配帧数: %d / %d (跟踪帧)\n', length(common_t), length(t_track));

    fprintf('\n【方位角误差】\n');
    fprintf('  均值: %.3f 度\n', mean(abs(err_az)));
    fprintf('  标准差: %.3f 度\n', std(err_az));
    fprintf('  最大误差: %.3f 度\n', max(abs(err_az)));
    fprintf('  RMSE: %.3f 度\n', sqrt(mean(err_az.^2)));
    
    fprintf('\n【俯仰角误差】\n');
    fprintf('  均值: %.3f 度\n', mean(abs(err_el)));
    fprintf('  标准差: %.3f 度\n', std(err_el));
    fprintf('  最大误差: %.3f 度\n', max(abs(err_el)));
    fprintf('  RMSE: %.3f 度\n', sqrt(mean(err_el.^2)));
    
    fprintf('\n【距离误差】\n');
    fprintf('  均值: %.1f m\n', mean(abs(err_r)));
    fprintf('  标准差: %.1f m\n', std(err_r));
    fprintf('  最大误差: %.1f m\n', max(abs(err_r)));
    fprintf('  RMSE: %.1f m\n', sqrt(mean(err_r.^2)));
    r_mean_meas = mean(r_meas_match);
    fprintf('  相对误差: %.2f %%\n', mean(abs(err_r))/r_mean_meas*100);
    
    fprintf('\n【速度误差】\n');
    fprintf('  均值: %.2f m/s\n', mean(abs(err_v)));
    fprintf('  标准差: %.2f m/s\n', std(err_v));
    fprintf('  最大误差: %.2f m/s\n', max(abs(err_v)));
    fprintf('  RMSE: %.2f m/s\n', sqrt(mean(err_v.^2)));
end

%% ====================== 球坐标转直角坐标（雷达站心坐标系） ======================
% 坐标系定义：
%   X轴：方位角0度方向（正北），X = R*cos(El)*cos(Az)
%   Y轴：方位角90度方向（正东），Y = R*cos(El)*sin(Az)
%   Z轴：天顶方向，Z = R*sin(El)
% 注意：方位角从X轴（北）逆时针为正

% 所有有效点（杂波）

% 跟踪结果
